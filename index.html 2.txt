<!-- Three.js hero dumbbell (self-contained, runs after DOM+THREE loaded) -->
<script>
(function initHero3D() {
  // Wait for THREE to be available (loaded with defer)
  function tryInit() {
    if (typeof THREE === 'undefined' || typeof gsap === 'undefined') {
      setTimeout(tryInit, 80);
      return;
    }

    const canvas = document.getElementById('db-canvas');
    if (!canvas) return;

    const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true });
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    renderer.setClearColor(0x000000, 0);

    const scene  = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(38, 1, 0.1, 50);
    camera.position.set(0, 0.4, 7);

    const dbGroup = new THREE.Group();
    scene.add(dbGroup);

    const matBar      = new THREE.MeshStandardMaterial({ color: 0x1a1a1a, metalness: 0.95, roughness: 0.18 });
    const matPlateInner = new THREE.MeshStandardMaterial({ color: 0x111111, metalness: 0.9,  roughness: 0.22 });
    const matPlateOuter = new THREE.MeshStandardMaterial({ color: 0x0d0d0d, metalness: 0.95, roughness: 0.15 });
    const matCollar   = new THREE.MeshStandardMaterial({ color: 0xc87a20, metalness: 0.88, roughness: 0.25 });
    const matKnurl    = new THREE.MeshStandardMaterial({ color: 0x222222, metalness: 0.8,  roughness: 0.55 });

    function cyl(rT, rB, h, mat, x=0, y=0, z=0, rotZ=0) {
      const geo  = new THREE.CylinderGeometry(rT, rB, h, 32);
      const mesh = new THREE.Mesh(geo, mat);
      mesh.position.set(x, y, z);
      mesh.rotation.z = rotZ;
      mesh.castShadow = true;
      dbGroup.add(mesh);
      return mesh;
    }

    function plate(x, r, halfThick, r2, mat) {
      const geo  = new THREE.CylinderGeometry(r, r, halfThick * 2, 48);
      const mesh = new THREE.Mesh(geo, mat);
      mesh.position.set(x, 0, 0);
      mesh.rotation.z = Math.PI / 2;
      mesh.castShadow = true;
      dbGroup.add(mesh);
    }

    // Bar
    cyl(0.09, 0.09, 4.8, matBar, 0, 0, 0, Math.PI/2);
    // Knurl zones
    cyl(0.095, 0.095, 0.9, matKnurl, -0.55, 0, 0, Math.PI/2);
    cyl(0.095, 0.095, 0.9, matKnurl,  0.55, 0, 0, Math.PI/2);
    // Collars (gold)
    cyl(0.115, 0.115, 0.18, matCollar, -1.28, 0, 0, Math.PI/2);
    cyl(0.115, 0.115, 0.18, matCollar,  1.28, 0, 0, Math.PI/2);
    // Thin inner plates
    plate(-1.72, 0.38, 0.10, 0.22, matPlateInner);
    plate( 1.50, 0.38, 0.10, 0.22, matPlateInner);
    // Main outer plates
    plate(-2.10, 0.62, 0.095, 0.34, matPlateOuter);
    plate( 1.76, 0.62, 0.095, 0.34, matPlateOuter);
    // End caps
    cyl(0.095, 0.095, 0.18, matCollar, -2.62, 0, 0, Math.PI/2);
    cyl(0.095, 0.095, 0.18, matCollar,  2.62, 0, 0, Math.PI/2);
    cyl(0.18,  0.18,  0.06, matCollar, -2.72, 0, 0, Math.PI/2);
    cyl(0.18,  0.18,  0.06, matCollar,  2.72, 0, 0, Math.PI/2);

    // Shadow plane
    const shadowPlane = new THREE.Mesh(
      new THREE.PlaneGeometry(6, 3),
      new THREE.ShadowMaterial({ opacity: 0.55 })
    );
    shadowPlane.rotation.x = -Math.PI / 2;
    shadowPlane.position.y = -1.05;
    shadowPlane.receiveShadow = true;
    scene.add(shadowPlane);

    // Lights
    const keyLight = new THREE.SpotLight(0xffffff, 3.5);
    keyLight.position.set(4, 5, 4);
    keyLight.angle = Math.PI / 7;
    keyLight.penumbra = 0.65;
    keyLight.decay = 1.8;
    keyLight.castShadow = true;
    keyLight.shadow.mapSize.width = keyLight.shadow.mapSize.height = 1024;
    scene.add(keyLight);
    scene.add(keyLight.target);

    const fillLight = new THREE.DirectionalLight(0x8ab4f8, 0.45);
    fillLight.position.set(-5, 2, 1);
    scene.add(fillLight);

    const rimLight = new THREE.SpotLight(0xffffff, 1.8);
    rimLight.position.set(-2, 3, -5);
    rimLight.angle = Math.PI / 6;
    rimLight.penumbra = 0.8;
    rimLight.decay = 2.0;
    scene.add(rimLight);

    scene.add(new THREE.AmbientLight(0x111111, 0.5));
    const groundGlow = new THREE.PointLight(0xc87a20, 0.35, 6);
    groundGlow.position.set(0, -1.5, 1.5);
    scene.add(groundGlow);

    // Particles
    const PARTICLE_COUNT = 80;
    const pPositions = new Float32Array(PARTICLE_COUNT * 3);
    const pSizes     = new Float32Array(PARTICLE_COUNT);
    for (let i = 0; i < PARTICLE_COUNT; i++) {
      pPositions[i*3]   = (Math.random()-.5)*10;
      pPositions[i*3+1] = (Math.random()-.5)*8;
      pPositions[i*3+2] = (Math.random()-.5)*6 - 1;
      pSizes[i] = Math.random()*2.5 + 0.8;
    }
    const pGeo = new THREE.BufferGeometry();
    pGeo.setAttribute('position', new THREE.BufferAttribute(pPositions, 3));
    pGeo.setAttribute('size',     new THREE.BufferAttribute(pSizes, 1));
    const particles = new THREE.Points(pGeo, new THREE.PointsMaterial({
      color: 0xffffff, size: 0.028, transparent: true, opacity: 0.18,
      sizeAttenuation: true, depthWrite: false,
    }));
    scene.add(particles);

    // Mouse influence
    let targetRotX = 0, targetRotY = 0, currentRotX = 0, currentRotY = 0;
    document.addEventListener('mousemove', e => {
      const rect = canvas.getBoundingClientRect();
      targetRotX = ((e.clientY - rect.top  - rect.height/2) / (rect.height/2)) * 0.22;
      targetRotY = ((e.clientX - rect.left - rect.width/2)  / (rect.width/2))  * 0.18;
    });

    // GSAP auto-spin + float
    const spin    = { y: 0 };
    const floatObj= { y: 0 };
    gsap.to(spin,     { y: Math.PI*2, duration: 9, repeat: -1, ease: 'none' });
    gsap.to(floatObj, { y: 0.18, duration: 2.6, repeat: -1, yoyo: true, ease: 'sine.inOut' });

    // Resize
    function onResize() {
      const pw = canvas.parentElement?.offsetWidth  || 480;
      const ph = canvas.parentElement?.offsetHeight || 480;
      renderer.setSize(pw, ph);
      camera.aspect = pw / ph;
      camera.updateProjectionMatrix();
    }
    window.addEventListener('resize', onResize);
    onResize();

    // Render loop
    const clock = new THREE.Clock();
    const shadowMat = shadowPlane.material;

    function animate() {
      requestAnimationFrame(animate);
      const t = clock.getElapsedTime();

      currentRotX += (targetRotX - currentRotX) * 0.055;
      currentRotY += (targetRotY - currentRotY) * 0.055;

      dbGroup.rotation.y = spin.y + currentRotY;
      dbGroup.rotation.x = Math.sin(t * 0.25) * 0.06 + currentRotX;
      dbGroup.rotation.z = Math.sin(t * 0.18) * 0.025;
      dbGroup.position.y = floatObj.y;

      const pPos = pGeo.attributes.position.array;
      for (let i = 0; i < PARTICLE_COUNT; i++) {
        pPos[i*3+1] += 0.0008;
        if (pPos[i*3+1] > 4) pPos[i*3+1] = -4;
      }
      pGeo.attributes.position.needsUpdate = true;

      keyLight.position.x = 4 + Math.sin(t * 0.3) * 0.8;
      keyLight.position.z = 4 + Math.cos(t * 0.25) * 0.6;

      const ss = 1 - floatObj.y * 0.3;
      shadowPlane.scale.set(ss, ss, 1);
      shadowMat.opacity = 0.55 - floatObj.y * 0.18;

      renderer.render(scene, camera);
    }
    animate();

    // Entrance
    gsap.from(dbGroup.scale, { x: 0, y: 0, z: 0, duration: 1.4, ease: 'elastic.out(1,0.7)', delay: 0.6 });
  }

  tryInit();
})();
</script>
